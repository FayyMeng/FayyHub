local function reportError(context, err)
    local now = os.clock()

        return
    end

end

local function formatShort(n, prefix)
    n = tonumber(n) or 0
    prefix = prefix or ""

    local sign = n < 0 and "-" or ""
    n = math.abs(n)

    if n < 1000 then
        return string.format("%s%s%d", sign, prefix, math.floor(n + 0.5))
    end

    local index = 0
    while n >= 1000 and index < #SUFFIXES - 1 do
        n /= 1000
        index += 1
    end

    return string.format("%s%s%.2f%s", sign, prefix, n, SUFFIXES[index + 1])
end

local function formatWeight(kg)
    kg = tonumber(kg) or 0
    if kg >= 1000 then
        return formatShort(kg) .. "kg"
    end
    return string.format("%.1fkg", kg)
end

local function formatDistance(studs)
    studs = tonumber(studs) or 0
    if studs >= 1000 then
        return string.format("%.1fkm", studs / 1000)
    end
    return string.format("%dm", math.floor(studs + 0.5))
end

local function formatLuck(score)
    local pct = (tonumber(score) or 0) * 100

    if pct <= 0 then
        return "+0%"
    end
    if pct < 1 then
        return string.format("+%.2f%%", pct)
    end
    if pct < 10 then
        return string.format("+%.1f%%", pct)
    end
    return string.format("+%.0f%%", pct)
end

local function parseValue(text)
    if type(text) ~= "string" then
        return nil
    end

    local cleaned = text:lower():gsub("[%s,%$_]", "")
    if cleaned == "" then
        return 0
    end

    local number, suffix = cleaned:match("^(%d*%.?%d+)(%a*)$")
    if not number then
        return nil
    end

    local base = tonumber(number)
    if not base then
        return nil
    end

    if suffix == "" then
        return base
    end

    local multiplier = PARSE_MULTIPLIERS[suffix]
    if not multiplier then
        return nil
    end

    return base * multiplier
end

local function mutationLuck(name)
    if type(name) ~= "string" or name == "" then
        return 1
    end

end

local function combinedLuckMult(inst)
    local mutation = getAttr(inst, "Mutation")
    local roll = tonumber(getAttr(inst, "MutationLuckRoll"))
    local multiplier = (roll and roll > 0) and roll or mutationLuck(mutation)

    local extra = getAttr(inst, "ExtraMutations")
    if type(extra) == "string" and extra ~= "" then
        for name in string.gmatch(extra, "[^,]+") do
            if name ~= "" then
                multiplier *= mutationLuck(name)
            end
        end
    end

    if getAttr(inst, "IsBloodCrystal") == true then
        multiplier *= LUCK.blood
    end

    if getAttr(inst, "AdminMutation") == "Radioactive" and mutation ~= "Radioactive" then
        local hasRadioactive = type(extra) == "string" and extra:find("Radioactive", 1, true) ~= nil
        if not hasRadioactive then
            multiplier *= mutationLuck("Radioactive")
        end
    end

    return multiplier
end

local function computeLuck(inst)
    local tier = crystalTier(inst)
    if tier <= 0 then
        return 0
    end

    local weight = math.max(0, crystalWeight(inst))
    local base = (LUCK.rarity[tier] or LUCK.rarity[1]) * math.min(weight, LUCK.cap) ^ LUCK.exponent * LUCK.base

    if getAttr(inst, "BombCrystal") == true then
        base *= LUCK.bomb
    end

    return base * combinedLuckMult(inst)
end

local function getOwnedPlotForPlace()
    local things = Workspace:FindFirstChild("Things")
    local plots = things and things:FindFirstChild("Plots")
    local slots = plots and plots:FindFirstChild("Slots")
    if not slots then
        return nil, nil
    end
    local plot = slots:FindFirstChild(LocalPlayer.Name)
    if not plot then
        for _, slot in ipairs(slots:GetChildren()) do
            local owner = slot:GetAttribute("Owner") or slot:GetAttribute("OwnerName")
            if owner == LocalPlayer.Name or owner == LocalPlayer.UserId or tostring(owner) == tostring(LocalPlayer.UserId) then
                plot = slot
                break
            end
        end
    end
    if not plot then
        return nil, nil
    end
    local region = plot:FindFirstChild("Region")
    if not (region and region:IsA("BasePart")) then
        return plot, nil
    end
    return plot, region
end

local function crystalPlaceWorldPos(region, objectSize)
    local cf = region.CFrame
    local size = region.Size
    local half = size * 0.5
    local padX = math.max(half.X - 1.5, 1)
    local padZ = math.max(half.Z - 1.5, 1)
    local oh = (objectSize or Vector3.new(2, 2, 2)) * 0.5
    local lx = (math.random() - 0.5) * 2 * padX
    local lz = (math.random() - 0.5) * 2 * padZ
    local ly = oh.Y + 0.05
    local bottomLocal = Vector3.new(lx, ly - half.Y, lz)
    return cf:PointToWorldSpace(bottomLocal)
end

local function toolPlaceSize(tool)
    local handle = tool and tool:FindFirstChild("Handle")
    if handle and handle:IsA("BasePart") then
        return handle.Size
    end
    local fx = tonumber(tool and getAttr(tool, "FinalSizeX"))
    local fy = tonumber(tool and getAttr(tool, "FinalSizeY"))
    local fz = tonumber(tool and getAttr(tool, "FinalSizeZ"))
    if fx and fy and fz then
        return Vector3.new(fx, fy, fz)
    end
    return Vector3.new(2, 2, 2)
end

local function isPlaceableCrystalTool(tool)
    if not tool or not tool:IsA("Tool") then
        return false
    end
    if getAttr(tool, "Tier") == nil then
        return false
    end
    -- runes often have Rune in name / no WeightKg sometimes - still allow Tier tools
    if type(tool.Name) == "string" and tool.Name:find("Rune", 1, true) then
        return false
    end
    return true
end

local function placeCrystalToolOnPlot(tool, remote, region)
    if not tool or not tool.Parent or not remote or not region then
        return false
    end
    local humanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
    if humanoid and tool.Parent ~= LocalPlayer.Character then
        pcall(function()
            humanoid:EquipTool(tool)
        end)
        task.wait(0.15)
    end
    local held = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild(tool.Name)
    held = held or tool
    if not held or not held.Parent then
        return false
    end
    local pos = crystalPlaceWorldPos(region, toolPlaceSize(held))
    local yaw = math.random() * math.pi * 2
    local ok = pcall(function()
        remote:FireServer(held.Name, pos, yaw, held)
    end)
    return ok == true
end

            if not container then return end
            for _, child in ipairs(container:GetChildren()) do

                    tools[#tools + 1] = child
                end
            end

        scan(LocalPlayer:FindFirstChildOfClass("Backpack"))

local function hasActiveRune(keyword)
    local data = LocalPlayer:FindFirstChild("PlayerData")
    local plot = data and data:FindFirstChild("PlotData")
    local runes = plot and plot:FindFirstChild("Runes")
    if not runes then
        return false
    end

    for _, child in ipairs(runes:GetChildren()) do
        local runeName = child:GetAttribute("RuneName")
        if type(runeName) == "string" and runeName:find(keyword, 1, true) then
            if (tonumber(child:GetAttribute("Remaining")) or 0) > 0 then
                return true
            end
        end
    end

    return false
end

local function backpackCapacity()
    if LocalPlayer:GetAttribute("InfBackpack") == true then
        return math.huge
    end

    local base = realStat("CarryWeight") or 10
    if ownsGamepass("CarryKgPlus4") then
        base *= 4
    end

    local total = base + (realStat("CarryWeightBonus") or 0)

    if hasActiveRune("Weight") then
        return total * 2
    end

    return total
end

local function backpackWeight()
    local total = 0

    local function scan(container)
        if not container then
            return
        end
        for _, child in ipairs(container:GetChildren()) do
            if child:IsA("Tool") and getAttr(child, "Tier") ~= nil then
                local kg = tonumber(getAttr(child, "WeightKg"))
                if kg then
                    total += kg
                end
            end
        end
    end

    scan(LocalPlayer:FindFirstChildOfClass("Backpack"))
    scan(LocalPlayer.Character)

    return total
end

local function backpackFree()
    local capacity = backpackCapacity()
    if capacity == math.huge then
        return math.huge
    end
    return capacity - backpackWeight()
end

local function looksLikeCrystal(inst)
    if not inst:IsA("BasePart") then
        return false
    end
    return inst.Name:find("Crystal", 1, true) ~= nil
end

local crystalFlags = setmetatable({}, { __mode = "k" })

local function isCrystal(inst)
    local cached = crystalFlags[inst]
    if cached ~= nil then
        return cached
    end

    local result = false

    if inst:IsA("BasePart") and getAttr(inst, "Value") ~= nil then
        result = getAttr(inst, "CrystalName") ~= nil or inst.Name:find("Crystal", 1, true) ~= nil
    end

    crystalFlags[inst] = result

    return result
end

local function rebuildContainers()
    table.clear(containerList)

    local seen = {}

    local function push(container)
        if not container or seen[container] then
            return
        end
        seen[container] = true
        containerList[#containerList + 1] = container
    end

    push(Workspace)

    for _, name in ipairs(CONTAINER_NAMES) do
        push(Workspace:FindFirstChild(name))
    end

    local things = Workspace:FindFirstChild("Things")
    if things then
        for _, name in ipairs(CONTAINER_NAMES) do
            push(things:FindFirstChild(name))
        end
    end
end

local function eachContainer(fn)
    local now = os.clock()
    local stale = #containerList == 0 or now - containerClock >= 1.0

    if not stale then
        for _, container in ipairs(containerList) do
            if not container.Parent and container ~= Workspace then
                stale = true
                break
            end
        end
    end

    if stale then
        containerClock = now
        rebuildContainers()
    end

    for _, container in ipairs(containerList) do
        fn(container)
    end
end

local function untrackCrystal(inst)
    local conns = S.registry[inst]
    if not conns then
        return
    end

    for _, connection in ipairs(conns) do
        connection:Disconnect()
    end

    S.registry[inst] = nil
    S.registryCount -= 1
    S.dirty[inst] = nil
    S.candidates[inst] = nil
    S.statsDirty = true

    destroyEntry(inst)
end

local function trackCrystal(inst)
    if S.registry[inst] then
        return
    end

    local conns = {}
    S.registry[inst] = conns
    S.registryCount += 1
    S.statsDirty = true

    local ok = pcall(function()
        conns[#conns + 1] = inst.Destroying:Connect(function()
            untrackCrystal(inst)
        end)

        conns[#conns + 1] = inst.AncestryChanged:Connect(function()
            if not inst:IsDescendantOf(Workspace) then
                untrackCrystal(inst)
            end
        end)

        for _, name in ipairs(WATCHED_ATTRIBUTES) do
            conns[#conns + 1] = inst:GetAttributeChangedSignal(name):Connect(function()
                markDirty(inst)
            end)
        end

        local label = luckLabel(inst)
        if label then
            conns[#conns + 1] = label:GetPropertyChangedSignal("Text"):Connect(function()
                markDirty(inst)
            end)
        end
    end)

    if not ok then
        untrackCrystal(inst)
        return
    end

    markDirty(inst)
end

local function syncCrystal(inst)
    S.dirty[inst] = nil

    if not S.registry[inst] then
        return
    end

    if not inst.Parent then
        untrackCrystal(inst)
        return
    end

    local entry = S.espCache[inst]
    local hidden = not S.espActive or getAttr(inst, "Collected") == true

    if not hidden then
        hidden = not meetsFilter(inst)
    end

    if hidden then
        if entry then
            destroyEntry(inst, entry)
        end
        return
    end

    if not entry then
        local built, result = pcall(createEntry, inst)
        if not built then
            reportError("billboard", result)
            return
        end

        entry = result
        S.espCache[inst] = entry
        S.espCount += 1
        S.statsDirty = true
    end

    local signature = crystalSignature(inst)
    if signature == entry.signature then
        return
    end

    local root = getRoot()
    local ok, err = pcall(applyDetails, inst, entry, root and root.Position or nil)
    if ok then
        entry.signature = signature
    else
        reportError("details", err)
    end
end

local sweepSeen = {}

local function sweep()
    local seen = sweepSeen
    table.clear(seen)

    eachContainer(function(container)
        for _, child in ipairs(container:GetChildren()) do
            if not seen[child] and isCrystal(child) then
                seen[child] = true
                if not S.registry[child] then
                    trackCrystal(child)
                end
            end
        end
    end)

    local stale
    for inst in pairs(S.registry) do
        if not seen[inst] then
            stale = stale or {}
            stale[#stale + 1] = inst
        end
    end

    if stale then
        for _, inst in ipairs(stale) do
            untrackCrystal(inst)
        end
    end
end

local lastDistanceOrigin

local function updateDistances()
    local root = getRoot()
    if not root then
        return
    end

    local origin = root.Position

    if lastDistanceOrigin and (origin - lastDistanceOrigin).Magnitude < 1 then
        return
    end

    lastDistanceOrigin = origin

    for inst, entry in pairs(S.espCache) do
        if inst.Parent then
            local text = formatDistance((inst.Position - origin).Magnitude)
            if text ~= entry.distanceText then
                applyExtra(entry, text)
            end
        end
    end
end

local function luckMeetsFilter(inst)
    if S.minLuck <= 0 then return true end
    local ok, luck = pcall(crystalLuck, inst)
    return (ok and luck or 0) * 100 >= S.minLuck
end

local function syncLuckCrystal(inst)
    if not S.luckEspActive or getAttr(inst, "Collected") == true or not luckMeetsFilter(inst) then
        destroyLuckEntry(inst)
        return
    end

    local entry = S.luckEspCache[inst]
    if not entry then
        local built, result = pcall(createLuckEntry, inst)
        if not built then return end
        entry = result
        S.luckEspCache[inst] = entry
        S.luckEspCount += 1
        S.statsDirty = true
    end

    local ok, luck = pcall(crystalLuck, inst)
    local luckText = formatLuck(ok and luck or 0)
    local root = getRoot()
    local distanceText = "--"
    if root then
        distanceText = formatDistance((inst.Position - root.Position).Magnitude)
    end

    local textSize = crystalTextSize()
    entry.gui.Size = luckEspGuiSize()
    if entry.constraint then
        entry.constraint.MaxTextSize = textSize
    end
    if entry.constraints then
        for _, c in ipairs(entry.constraints) do
            c.MaxTextSize = textSize
        end
    end

    if entry.luckText ~= luckText then
        entry.luckText = luckText
        entry.label.Text = luckText
        entry.label.TextColor3 = Color3.fromRGB(255, 196, 0)
    end
    if entry.distanceText ~= distanceText then
        entry.distanceText = distanceText
        if entry.distLabel then
            entry.distLabel.Text = distanceText
            entry.distLabel.TextColor3 = COLORS.extra
        end
    end
end

    for container, connection in pairs(S.containerConns) do

local function updateTracking()
    if trackingEnabled() then
        S.sweepAccumulator = math.huge
        watchContainers()
        requestRefresh()

end

local espConn = RunService.Heartbeat:Connect(function(deltaTime)
    if S.statsDirty and S.StatsLabel then
        S.statsDirty = false
        S.StatsLabel:SetText(string.format("Tracking: %d  |  Shown: %d", S.registryCount, S.espCount))
    end

    if not trackingEnabled() then
        return
    end

    local now = os.clock()

    if next(S.candidates) ~= nil then
        for inst, expiry in pairs(S.candidates) do
            if not inst.Parent then
                S.candidates[inst] = nil
            elseif isCrystal(inst) then
                S.candidates[inst] = nil
                trackCrystal(inst)
            elseif now > expiry then
                S.candidates[inst] = nil
            end
        end
    end

    if next(S.dirty) ~= nil then
        local deadline = now + ESP.budget
        local processed = 0
        for inst in pairs(S.dirty) do
            local ok, err = pcall(syncCrystal, inst)
            if not ok then
                S.dirty[inst] = nil
                reportError("sync", err)
            end
            processed += 1
            if processed >= 12 or os.clock() > deadline then
                break
            end
        end
    end

    S.sweepAccumulator += deltaTime
    if S.sweepAccumulator >= ESP.sweep then
        S.sweepAccumulator = 0
        local ok, err = pcall(function()
            watchContainers()
            sweep()
            if S.luckEspActive then
                refreshLuckEsp()
            end
        end)
        if not ok then
            reportError("sweep", err)
        end
    end

    if S.espCount > 0 then
        S.distanceAccumulator += deltaTime
        if S.distanceAccumulator >= PACE.distance then
            S.distanceAccumulator = 0
            local ok, err = pcall(updateDistances)
            if not ok then
                reportError("distance", err)
            end
        end
    end
end)

    for _, player in ipairs(Players:GetPlayers()) do

                            entry.highlight.Adornee = character

            gone[#gone + 1] = player
        end

        end
    end

local streamMark = 0
local streamSpot

local function requestStream(position)
    if typeof(position) ~= "Vector3" then
        return
    end

    local now = os.clock()

    if streamSpot and now - streamMark < 0.3 and (streamSpot - position).Magnitude < 32 then
        return
    end

    streamMark = now
    streamSpot = position

    task.spawn(function()
        pcall(function()
            LocalPlayer:RequestStreamAroundAsync(position, 1)
        end)
    end)
end

local function applyPivot(cframe)
    local character = LocalPlayer.Character
    if not character then
        return false
    end

    requestStream(cframe.Position)

    local root = getRoot()
    if not root then
        return false
    end

    local moved = pcall(function()
        character:PivotTo(cframe)
    end)

    if not moved then
        moved = pcall(function()
            root.CFrame = cframe
        end)
    end

    if not moved then
        return false
    end

    pcall(function()
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end)

    return true
end

local function findClearGoal(position, ignore)
    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = ignore
    params.MaxParts = 1

    for _, offset in ipairs(TP.clear) do
        local candidate = position + offset
        local ok, hits = pcall(function()
            return Workspace:GetPartBoundsInRadius(candidate, 2.5, params)
        end)

        if ok and #hits == 0 then
            return candidate
        end
    end

    return position + TP.clear[#TP.clear]
end

local function finishTeleport()
    if not S.tpState then
        return
    end

    local root = getRoot()
    if root then
        pcall(function()
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end)
    end

    local character = LocalPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        pcall(function()
            humanoid.PlatformStand = false
            humanoid:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
            humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
            humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true)
            humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        end)
    end

    S.tpState = nil
end

local function teleportTo(target)
    local position
    if typeof(target) == "Vector3" then
        position = target
    elseif typeof(target) == "Instance" and target.Parent then
        position = target.Position
    end

    if not position then
        return false
    end

    local character = LocalPlayer.Character
    if not character then
        return false
    end

    local root = getRoot()
    if not root then
        return false
    end

    finishTeleport()
    if Move and Move.glideStop then
        pcall(Move.glideStop)
    end

    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        pcall(function()
            if humanoid.SeatPart or humanoid.Sit then
                humanoid.Sit = false
            end
            humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
            humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
        end)
    end

    local ignore = { character }
    if typeof(target) == "Instance" then
        ignore[#ignore + 1] = target
    end

    local goalPos = findClearGoal(position + TP.offset, ignore)
    local goalFrame = CFrame.new(goalPos, goalPos + Vector3.new(0, 0, -1))

    if Move and Move.glide then
        local ok = Move.glide(goalFrame, position)
        if ok then
            S.tpState = {
                goal = goalFrame,
                holdUntil = os.clock() + 2.5,
                usingGlide = true,
            }
            task.spawn(function()
                local deadline = os.clock() + 4
                while os.clock() < deadline do
                    local r = getRoot()
                    if r and (r.Position - goalPos).Magnitude <= 6 then
                        break
                    end
                    if Move.glide then
                        pcall(Move.glide, goalFrame, position)
                    end
                    task.wait(0.05)
                end
                if Move.glideStop then
                    pcall(Move.glideStop)
                end
                applyPivot(goalFrame)
                finishTeleport()
            end)
            return true
        end
    end

    if not applyPivot(goalFrame) then
        return false
    end

    S.tpState = {
        goal = goalFrame,
        holdUntil = os.clock() + TP.hold,
    }

    return true
end

local promptCache = setmetatable({}, { __mode = "k" })

local function crystalPrompt(inst)
    local cached = promptCache[inst]
    if cached and cached.Parent then
        return cached
    end

    local ok, prompt = pcall(inst.FindFirstChildOfClass, inst, "ProximityPrompt")

    if not (ok and prompt) then
        ok, prompt = pcall(inst.FindFirstChildWhichIsA, inst, "ProximityPrompt", true)
    end

    if ok and prompt then
        promptCache[inst] = prompt
        return prompt
    end

    promptCache[inst] = nil

    return nil
end

local function surfaceDistance(part, origin)
    local ok, distance = pcall(function()
        local point = part.CFrame:PointToObjectSpace(origin)
        local half = part.Size * 0.5
        local clamped = Vector3.new(
            math.clamp(point.X, -half.X, half.X),
            math.clamp(point.Y, -half.Y, half.Y),
            math.clamp(point.Z, -half.Z, half.Z)
        )
        return (point - clamped).Magnitude
    end)

    if ok and distance then
        return distance
    end

    return (part.Position - origin).Magnitude
end

local function firePrompt(prompt)
    if not promptRestores[prompt] then
        promptRestores[prompt] = {
            hold = prompt.HoldDuration,
            sight = prompt.RequiresLineOfSight,
            enabled = prompt.Enabled,
            range = prompt.MaxActivationDistance,
        }
    end

    pcall(function()
        prompt.HoldDuration = 0
        prompt.RequiresLineOfSight = false
        prompt.Enabled = true
        prompt.MaxActivationDistance = 1000
    end)

    local fired = false

    if typeof(fireproximityprompt) == "function" then
        fired = pcall(fireproximityprompt, prompt, 1)
        if not fired then
            fired = pcall(fireproximityprompt, prompt)
        end
    end

    if not fired then
        fired = pcall(function()
            prompt:InputHoldBegin()
            prompt:InputHoldEnd()
        end)
    end

    schedule(PICK.restore, function()
        local saved = promptRestores[prompt]
        if not saved then
            return
        end

        promptRestores[prompt] = nil

        if prompt.Parent then
            prompt.HoldDuration = saved.hold
            prompt.RequiresLineOfSight = saved.sight
            prompt.Enabled = saved.enabled
            prompt.MaxActivationDistance = saved.range
        end
    end)

    return fired
end

    pickupParams.RespectCanCollide = false

local pickupFound = {}
local pickupSeen = {}

local function pickupCandidates(free, origin)
    local found = pickupFound
    local seen = pickupSeen
    table.clear(found)
    table.clear(seen)
    local now = os.clock()

    local function consider(child)
        if not child or seen[child] then
            return
        end
        seen[child] = true

        if not child.Parent or not isCrystal(child) or getAttr(child, "Collected") == true then
            return
        end

        local claim = claimed[child]
        if claim and now - claim < PICK.retry then
            return
        end

        local value = crystalValue(child)

        if S.autoPickupActive and not S.pickupValueActive and not S.pickupLuckActive then
            if not meetsFilter(child, value) then
                return
            end
        end

        if S.pickupValueActive then
            if value < (S.pickupMinValue or 0) then
                return
            end
        end

        if S.pickupLuckActive then
            local ok, luck = pcall(crystalLuck, child)
            local luckPct = (ok and luck or 0) * 100
            if luckPct < (S.pickupMinLuck or 0) then
                return
            end
        end

        if not S.autoPickupActive and not S.pickupValueActive and not S.pickupLuckActive then
            return
        end

        local weight = crystalWeight(child)
        if weight > free then
            return
        end

        local distance = surfaceDistance(child, origin)
        if distance > PICK.range then
            return
        end

        found[#found + 1] = {
            inst = child,
            prompt = crystalPrompt(child),
            value = value,
            weight = weight,
            distance = distance,
        }
    end

    pickupParams.FilterDescendantsInstances = { LocalPlayer.Character or LocalPlayer }

    local ok, hits = pcall(function()
        return Workspace:GetPartBoundsInRadius(origin, PICK.range + PICK.pad, pickupParams)
    end)

    if ok and hits then
        for _, part in ipairs(hits) do
            consider(part)
        end
    end

    eachContainer(function(container)
        for _, child in ipairs(container:GetChildren()) do
            if child:IsA("BasePart") then
                consider(child)
            elseif child:IsA("Model") then
                for _, inner in ipairs(child:GetChildren()) do
                    consider(inner)
                end
            end
        end
    end)

    for inst in pairs(S.registry) do
        consider(inst)
    end

    table.sort(found, function(a, b)
        if a.value == b.value then
            return a.distance < b.distance
        end
        return a.value > b.value
    end)

    return found
end

local function grabCrystal(inst, prompt)
    local sent = false

    if HoldComplete then
        sent = pcall(function()
            HoldComplete:FireServer(inst)
        end)
    end

    if not prompt then
        prompt = crystalPrompt(inst)
    end

    if prompt and prompt.Parent and firePrompt(prompt) then
        sent = true
    end

    if not sent and typeof(fireclickdetector) == "function" then
        local ok, detector = pcall(inst.FindFirstChildWhichIsA, inst, "ClickDetector", true)
        if ok and detector then
            sent = pcall(fireclickdetector, detector, 0)
        end
    end

    return sent
end

local function instantPromptPatch(prompt)
    if S.instantPatched[prompt] or promptRestores[prompt] then
        return
    end

    S.instantPatched[prompt] = {
        hold = prompt.HoldDuration,
        sight = prompt.RequiresLineOfSight,
        enabled = prompt.Enabled,
    }

    pcall(function()
        prompt.HoldDuration = 0
        prompt.RequiresLineOfSight = false
        prompt.Enabled = true
    end)
end

local function restoreInstantPrompts()
    for prompt, saved in pairs(S.instantPatched) do
        if prompt.Parent then
            pcall(function()
                prompt.HoldDuration = saved.hold
                prompt.RequiresLineOfSight = saved.sight
                prompt.Enabled = saved.enabled
            end)
        end
    end

    table.clear(S.instantPatched)
end

local function isRuneLike(inst)
    if not inst then
        return false
    end
    if getAttr(inst, "IsRune") == true then
        return true
    end
    if getAttr(inst, "RuneId") ~= nil or getAttr(inst, "RuneName") ~= nil then
        return true
    end
    if type(inst.Name) == "string" and inst.Name:find("Rune", 1, true) then
        return true
    end
    return false
end

local function refreshInstantPrompts()
    local root = getRoot()
    if not root then
        return
    end

    for prompt in pairs(S.instantPatched) do
        if not prompt.Parent then
            S.instantPatched[prompt] = nil
        end
    end

    -- Patch ALL nearby ProximityPrompts (crystals, runes, shops, etc.)
    local radius = PICK.instantRadius or 90
    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { LocalPlayer.Character or LocalPlayer }
    pcall(function()
        params.MaxParts = 80
    end)

    local ok, parts = pcall(function()
        return Workspace:GetPartBoundsInRadius(root.Position, radius, params)
    end)

    local seen = {}
    if ok and parts then
        for _, part in ipairs(parts) do
            if part and part.Parent then
                for _, inst in ipairs(part:GetChildren()) do
                    if inst:IsA("ProximityPrompt") and not seen[inst] then
                        seen[inst] = true
                        instantPromptPatch(inst)
                    end
                end
                local parent = part.Parent
                if parent and parent ~= Workspace then
                    for _, inst in ipairs(parent:GetChildren()) do
                        if inst:IsA("ProximityPrompt") and not seen[inst] then
                            seen[inst] = true
                            instantPromptPatch(inst)
                        end
                    end
                end
            end
        end
    end

end


local function isBackpackCrystalTool(tool)
    if not tool or not tool:IsA("Tool") then
        return false
    end
    -- crystals have Tier + WeightKg; exclude pure runes-only if no Tier
    if getAttr(tool, "Tier") ~= nil and getAttr(tool, "WeightKg") ~= nil then
        return true
    end
    if getAttr(tool, "Value") ~= nil and getAttr(tool, "WeightKg") ~= nil then
        return true
    end
    return false
end

local function collectBackpackCrystals()
    local list = {}
    local function scan(container)
        if not container then
            return
        end
        for _, child in ipairs(container:GetChildren()) do
            if isBackpackCrystalTool(child) then
                list[#list + 1] = child
            end
        end
    end
    scan(LocalPlayer:FindFirstChildOfClass("Backpack"))
    scan(LocalPlayer.Character)
    return list
end

local function dropAllCrystals()
    local remote = findRemote("CrystalDropRequest")
    if not remote then
        WindUI:Notify({
            Title = "Drop Crystals",
            Content = "CrystalDropRequest missing",
            Duration = 3,
            Icon = "alert-circle",
        })
        return
    end

    local crystals = collectBackpackCrystals()
    if #crystals == 0 then
        WindUI:Notify({
            Title = "Drop Crystals",
            Content = "No crystals in backpack",
            Duration = 2,
            Icon = "alert-circle",
        })
        return
    end

    if S._droppingCrystals then
        return
    end
    S._droppingCrystals = true
    S._stopCrystalOps = false

    task.spawn(function()
        local humanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        local dropped = 0

        for _, tool in ipairs(crystals) do
            if S._stopCrystalOps then
                break
            end
            if not tool or not tool.Parent then
                continue
            end

            if humanoid and tool.Parent ~= LocalPlayer.Character then
                pcall(function()
                    humanoid:EquipTool(tool)
                end)
                task.wait(0.05)
            end

            local held = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild(tool.Name)
            local name = (held and held.Name) or tool.Name

            local ok = pcall(function()
                remote:FireServer(name)
            end)
            if ok then
                dropped += 1
            end
            task.wait(0.08)
        end

        S._droppingCrystals = false
        WindUI:Notify({
            Title = "Drop Crystals",
            Content = string.format("Dropped %d", dropped),
            Duration = 2,
            Icon = dropped > 0 and "check" or "alert-circle",
        })
    end)
end

local function pickupAllDroppedCrystals()
    local root = getRoot()
    if not root then
        return
    end

    -- Player-dropped crystals can still be picked even when backpack is full
    if S._pickingDropped then
        return
    end
    S._pickingDropped = true
    S._stopCrystalOps = false

    task.spawn(function()
        local radius = 80
        local params = OverlapParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = { LocalPlayer.Character or LocalPlayer }
        pcall(function()
            params.MaxParts = 100
        end)

        local ok, parts = pcall(function()
            return Workspace:GetPartBoundsInRadius(root.Position, radius, params)
        end)

        local targets = {}
        local seen = {}
        if ok and parts then
            for _, part in ipairs(parts) do
                if isCrystal(part) and getAttr(part, "Collected") ~= true and not seen[part] then
                    seen[part] = true
                    targets[#targets + 1] = part
                end
            end
        end

        -- also scan common drop folders
        for _, folderName in ipairs({ "DroppedCrystals", "Crystals", "Things" }) do
            local folder = Workspace:FindFirstChild(folderName)
            if folder then
                for _, child in ipairs(folder:GetDescendants()) do
                    if isCrystal(child) and getAttr(child, "Collected") ~= true and not seen[child] then
                        local pos = child:IsA("BasePart") and child.Position
                        if pos and (pos - root.Position).Magnitude <= radius then
                            seen[child] = true
                            targets[#targets + 1] = child
                        end
                    end
                end
            end
        end

        local picked = 0
        for _, crystal in ipairs(targets) do
            if S._stopCrystalOps then
                break
            end
            if not crystal.Parent or getAttr(crystal, "Collected") == true then
                continue
            end

            -- No teleport (less suspicious) — only fire nearby prompts
            local prompt = crystalPrompt(crystal)
            if prompt then
                instantPromptPatch(prompt)
            end
            if grabCrystal and grabCrystal(crystal, prompt) then
                picked += 1
            elseif prompt and firePrompt(prompt) then
                picked += 1
            end
            task.wait(1.2)
        end

        S._pickingDropped = false
        WindUI:Notify({
            Title = "Pickup Crystals",
            Content = string.format("Picked %d nearby", picked),
            Duration = 2,
            Icon = picked > 0 and "check" or "alert-circle",
        })
    end)
end

local function instantGrab()
    if not S.instantPromptActive then
        return
    end

    if backpackFree() <= 0 then
        return
    end

    local root = getRoot()
    if not root then
        return
    end

    local hits = nearbyCrystalParts(root.Position, PICK.range + PICK.pad)
    if not hits then
        return
    end

    local best, bestPrompt, bestDistance

    for _, part in ipairs(hits) do
        if part.Parent and isCrystal(part) and getAttr(part, "Collected") ~= true then
            local distance = surfaceDistance(part, root.Position)
            if distance <= PICK.range and (not best or distance < bestDistance) then
                best = part
                bestPrompt = crystalPrompt(part)
                bestDistance = distance
            end
        end
    end

    if not best then
        return
    end

    if bestPrompt then
        instantPromptPatch(bestPrompt)
    end

    if grabCrystal(best, bestPrompt) then
        claimed[best] = os.clock()
    end
end

local function pickupStep()
    local now = os.clock()
    if now - S.lastPickup < PICK.cooldown then
        return
    end

    local root = getRoot()
    if not root then
        return
    end

    local free = backpackFree()
    if free <= 0 then
        S.lastPickup = now
        table.clear(S.candidates)
        if now - S.lastBagWarn >= 12 then
            S.lastBagWarn = now
            WindUI:Notify({
                Title = "FayyMeng",
                Content = "Backpack full - pickup paused",
                Duration = 2,
                Icon = "alert-circle",
            })
        end
        return
    end

    for inst, stamp in pairs(claimed) do
        if now - stamp >= PICK.forget or not inst.Parent then
            claimed[inst] = nil
        end
    end

    S.candidates = pickupCandidates(free, root.Position)
    if #S.candidates == 0 then
        if free > 0.5 then
            requestStream(root.Position)
        end
        return
    end

    local budget = free
    local grabs = 0

    for _, entry in ipairs(S.candidates) do
        if grabs >= PICK.burst then
            break
        end

        if entry.weight <= budget and budget > 0 then
            claimed[entry.inst] = now

            if grabCrystal(entry.inst, entry.prompt) then
                budget -= entry.weight
                grabs += 1
            end
        end
    end

    if grabs > 0 then
        S.lastPickup = now
    else
        S.lastPickup = now
    end
end

local function tryFilteredCrystalPickup(origin, free)
    if free == nil then
        free = backpackFree()
    end
    if free <= 0 then
        return false
    end

    if not S.pickupValueActive and not S.pickupLuckActive then
        return false
    end

    local now = os.clock()
    -- Short gap between individual grabs (responsive, not instant-spam)
    if now - (S.lastFilteredPickup or 0) < 0.1 then
        return false
    end

    -- Heavy world scan is cached; grabs use the cache only
    local list = S._filterCandCache
    if not list or now - (S._filterCandAt or 0) > 0.65 then
        list = pickupCandidates(free, origin)
        -- Pre-filter by value/luck once when building cache
        local filtered = {}
        for _, entry in ipairs(list or {}) do
            local okValue = not S.pickupValueActive or entry.value >= (S.pickupMinValue or 0)
            local okLuck = true
            if S.pickupLuckActive then
                local luckOk, luck = pcall(crystalLuck, entry.inst)
                local luckPct = (luckOk and luck or 0) * 100
                okLuck = luckPct >= (S.pickupMinLuck or 0)
            end
            if okValue and okLuck and entry.weight <= free then
                filtered[#filtered + 1] = entry
            end
        end
        list = filtered
        S._filterCandCache = list
        S._filterCandAt = now
    end

    if not list or #list == 0 then
        return false
    end

    -- Take first still-valid candidate from cache
    for i = 1, #list do
        local entry = list[i]
        if entry and entry.inst and entry.inst.Parent and getAttr(entry.inst, "Collected") ~= true then
            claimed[entry.inst] = now
            if grabCrystal(entry.inst, entry.prompt) then
                S.lastFilteredPickup = now
                S.lastPickup = now
                table.remove(list, i)
                return true
            end
        end
    end

    -- Cache exhausted / stale
    S._filterCandCache = nil
    return false
end

local function updateBackpackLabel()
    if not S.BackpackLabel then
        return
    end

    local capacity = backpackCapacity()
    local used = backpackWeight()

    if capacity == math.huge then
        S.BackpackLabel:SetText(string.format("Bag %.1f / \u{221E} kg", used))
        return
    end

    local free = math.max(0, capacity - used)
    S.BackpackLabel:SetText(string.format("Bag %.1f / %.1f kg\nFree %.1f kg", used, capacity, free))
end

local schedulerConn = RunService.Heartbeat:Connect(function(deltaTime)
    if S.tpState then
        local ok, err = pcall(function()
            if not S.tpState.usingGlide then
                if not applyPivot(S.tpState.goal) then
                    finishTeleport()
                    return
                end
            end

            if os.clock() >= S.tpState.holdUntil then
                if S.tpState.usingGlide and Move and Move.glideStop then
                    pcall(Move.glideStop)
                end
                if S.tpState.goal then
                    applyPivot(S.tpState.goal)
                end
                finishTeleport()
            end
        end)

        if not ok then
            finishTeleport()
            reportError("teleport", err)
        end
    end

    if S.autoPickupActive or S.pickupValueActive or S.pickupLuckActive then
        S._pickupAcc = (S._pickupAcc or 0) + deltaTime
        if S._pickupAcc >= (PACE.pickup or 0.08) then
            S._pickupAcc = 0
            local ok, err = pcall(pickupStep)
            if not ok then
                reportError("pickup", err)
            end
        end
    end

    if S.instantPromptActive then
        S.instantAccumulator += deltaTime
        if S.instantAccumulator >= PICK.instantTick then
            S.instantAccumulator = 0
            local ok, err = pcall(refreshInstantPrompts)
            if not ok then
                reportError("instant", err)
            end
        end
    end

    if S.speedActive then
        local body = LocalPlayer.Character
        local mover = body and body:FindFirstChildOfClass("Humanoid")
        if mover then
            watchSpeed(mover)
            enforceSpeed(mover)
        end
    end

    if S.playerEspActive then
        S._playerEspAcc = (S._playerEspAcc or 0) + deltaTime
        if S._playerEspAcc >= (PACE.playerEsp or 0.25) then
            S._playerEspAcc = 0
            local ok, err = pcall(updatePlayerEsp)
            if not ok then
                reportError("playerEsp", err)
            end
        end
    end

    S.statsAccumulator += deltaTime
    if S.statsAccumulator >= PACE.stats then
        S.statsAccumulator = 0
        local ok, err = pcall(updateBackpackLabel)
        if not ok then
            reportError("backpack", err)
        end
    end

    if #pendingActions == 0 then
        return
    end

    local now = os.clock()
    for index = #pendingActions, 1, -1 do
        local job = pendingActions[index]
        if now >= job.at then
            table.remove(pendingActions, index)
            local ok, err = pcall(job.fn)
            if not ok then
                reportError("action", err)
            end
        end
    end
end)

local function sortedByScore(scoreFn)
    local scored = {}
    local seen = {}

    local function consider(inst)
        if seen[inst] or not inst.Parent then
            return
        end

        if getAttr(inst, "Collected") == true then
            return
        end

        seen[inst] = true

        local ok, score = pcall(scoreFn, inst)
        scored[#scored + 1] = { inst = inst, score = ok and score or 0 }
    end

    for inst in pairs(S.registry) do
        consider(inst)
    end

    eachContainer(function(container)
        for _, child in ipairs(container:GetChildren()) do
            if isCrystal(child) then
                consider(child)
            end
        end
    end)

    table.sort(scored, function(a, b)
        return a.score > b.score
    end)

    return scored
end

local aimParams = RaycastParams.new()
aimParams.FilterType = Enum.RaycastFilterType.Exclude
aimParams.IgnoreWater = true

local function getAimedCrystal()
    local unitRay = Mouse.UnitRay
    local origin = unitRay.Origin
    local direction = unitRay.Direction.Unit

    local character = LocalPlayer.Character
    aimParams.FilterDescendantsInstances = character and { character } or {}

    local hit = Workspace:Raycast(origin, direction * PICK.aimRange, aimParams)
    if hit and hit.Instance and (S.registry[hit.Instance] or isCrystal(hit.Instance)) then
        return hit.Instance
    end

    local best, bestDot
    local seen = {}

    local function consider(inst)
        if seen[inst] or not inst.Parent then
            return
        end

        seen[inst] = true

        local offset = (inst.Position + ESP.offset) - origin
        local magnitude = offset.Magnitude
        if magnitude > 0 then
            local dot = direction:Dot(offset / magnitude)
            if not bestDot or dot > bestDot then
                bestDot = dot
                best = inst
            end
        end
    end

    for inst in pairs(S.espCache) do
        consider(inst)
    end

    eachContainer(function(container)
        for _, child in ipairs(container:GetChildren()) do
            if isCrystal(child) then
                consider(child)
            end
        end
    end)

    if best and bestDot and bestDot >= PICK.aimDot then
        return best
    end
    return nil
end

            Duration = 3,
            Icon = "alert-circle",
        })
        return
    end

local function fireRemote(remote, ...)
    if not remote then
        return false
    end

    local args = table.pack(...)
    local ok = pcall(function()
        remote:FireServer(table.unpack(args, 1, args.n))
    end)

    return ok
end

local function unfavoriteAll()
    local cleared = 0

    local function scan(container)
        if not container then
            return
        end

        for _, child in ipairs(container:GetChildren()) do
            if child:IsA("Tool") and child:GetAttribute("Favorited") == true then
                pcall(function()
                    child:SetAttribute("Favorited", false)
                end)
                fireRemote(ToggleFavorite, child, false)
                cleared += 1
            end
        end
    end

    scan(LocalPlayer:FindFirstChildOfClass("Backpack"))
    scan(LocalPlayer.Character)

    return cleared
end

local function mountainSpot()
    local centerX = Workspace:GetAttribute("MountainCenterX")
    local centerZ = Workspace:GetAttribute("MountainCenterZ")

    if typeof(centerX) == "number" and typeof(centerZ) == "number" then
        local base = Workspace:GetAttribute("MountainBaseY")
        local peak = Workspace:GetAttribute("MountainPeakY")
        local height = 700

        if typeof(base) == "number" and typeof(peak) == "number" then
            height = base + (peak - base) * 0.55
        end

        return Vector3.new(centerX, height, centerZ)
    end

    local things = Workspace:FindFirstChild("Things")
    local zones = things and things:FindFirstChild("MountainZones")

    if zones then
        for _, child in ipairs(zones:GetChildren()) do
            if child:IsA("BasePart") and child.Name == "MountainZone" then
                return child.Position
            end
        end
    end

    return nil
end

local function mountainSpan()
    local radius = Workspace:GetAttribute("MountainRadius")

    if typeof(radius) == "number" and radius > 20 then
        return radius
    end

    return 150
end

local function isPointInPart(point, part)
    if not (part and part.Parent and part:IsA("BasePart")) then
        return false
    end
    local localPos = part.CFrame:PointToObjectSpace(point)
    local half = part.Size * 0.5
    return math.abs(localPos.X) <= half.X
        and math.abs(localPos.Y) <= half.Y
        and math.abs(localPos.Z) <= half.Z
end

local function distanceToPart(point, part)
    if not (part and part.Parent and part:IsA("BasePart")) then
        return math.huge
    end
    local localPos = part.CFrame:PointToObjectSpace(point)
    local half = part.Size * 0.5
    local dx = math.max(math.abs(localPos.X) - half.X, 0)
    local dy = math.max(math.abs(localPos.Y) - half.Y, 0)
    local dz = math.max(math.abs(localPos.Z) - half.Z, 0)
    return math.sqrt(dx * dx + dy * dy + dz * dz)
end

local function isInMountainZone(point)
    if typeof(point) ~= "Vector3" then
        return false
    end

    local cx = Workspace:GetAttribute("MountainCenterX")
    local cz = Workspace:GetAttribute("MountainCenterZ")
    local radius = Workspace:GetAttribute("MountainRadius")
    local baseY = Workspace:GetAttribute("MountainBaseY")
    local peakY = Workspace:GetAttribute("MountainPeakY")

    if typeof(cx) == "number" and typeof(cz) == "number" and typeof(radius) == "number"
        and typeof(baseY) == "number" and typeof(peakY) == "number" then
        local dx = point.X - cx
        local dz = point.Z - cz
        local dist = math.sqrt(dx * dx + dz * dz)
        return dist <= (radius + 60)
            and point.Y >= (baseY - 120)
            and point.Y <= (peakY + 120)
    end

    local things = Workspace:FindFirstChild("Things")
    local zones = things and things:FindFirstChild("MountainZones")
    if zones then
        for _, child in ipairs(zones:GetChildren()) do
            if child:IsA("BasePart") and child.Name == "MountainZone" and isPointInPart(point, child) then
                return true
            end
        end
    end

    return false
end

local function isInNoDigZone(point)
    local things = Workspace:FindFirstChild("Things")
    if not things then
        return false
    end

    local noDig = things:FindFirstChild("NoDigging")
    if noDig then
        if noDig:IsA("BasePart") and isPointInPart(point, noDig) then
            return true
        end
        for _, child in ipairs(noDig:GetChildren()) do
            if child:IsA("BasePart") and isPointInPart(point, child) then
                return true
            end
        end
    end

    local zones = things:FindFirstChild("Zones")
    if zones then
        for _, child in ipairs(zones:GetChildren()) do
            if child:IsA("BasePart") and child.Name == "NoDigZone" and isPointInPart(point, child) then
                return true
            end
        end
    end

    return false
end

local function isAtBase()
    local root = getRoot()
    if not root then
        return true
    end

    local pos = root.Position

    if isInMountainZone(pos) then
        S.wasAtBase = false
        return false
    end

    local spot = mountainSpot()
    if not spot then
        S.wasAtBase = pos.Y < 80
        return S.wasAtBase
    end

    local flat = Vector2.new(pos.X - spot.X, pos.Z - spot.Z)
    local dist = flat.Magnitude
    local span = mountainSpan()

    if S.wasAtBase then
        S.wasAtBase = dist > (span + 20)
    else
        S.wasAtBase = dist > (span + 55)
    end

    return S.wasAtBase == true
end

    unfavoriteAll()

local Move = {}

do
    local function install()
        local FLY_KEYS = {
            { key = Enum.KeyCode.W, axis = "look", sign = 1 },
            { key = Enum.KeyCode.S, axis = "look", sign = -1 },
            { key = Enum.KeyCode.D, axis = "right", sign = 1 },
            { key = Enum.KeyCode.A, axis = "right", sign = -1 },
            { key = Enum.KeyCode.Space, axis = "up", sign = 1 },
            { key = Enum.KeyCode.LeftControl, axis = "up", sign = -1 },
        }

        local flyActive = false
        local noclipActive = false
        local jumpActive = false
        local flySpeed = 100

        local velocity
        local gyro
        local flyConn
        local noclipConn
        local jumpConn
        local collisions = {}

        local function humanoidOf()
            local character = LocalPlayer.Character
            return character and character:FindFirstChildOfClass("Humanoid")
        end

        local function dropMovers()
            if velocity then
                pcall(function()
                    velocity:Destroy()
                end)
                velocity = nil
            end

            if gyro then
                pcall(function()
                    gyro:Destroy()
                end)
                gyro = nil
            end
        end

        local function attach(root)
            dropMovers()

            local ok = pcall(function()
                local body = Instance.new("BodyVelocity")
                body.Name = "FayyMengFlyVelocity"
                body.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                body.P = 9e4
                body.Velocity = Vector3.zero
                body.Parent = root
                velocity = body

                local spin = Instance.new("BodyGyro")
                spin.Name = "FayyMengFlyGyro"
                spin.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
                spin.P = 9e4
                spin.D = 500
                spin.CFrame = root.CFrame
                spin.Parent = root
                gyro = spin
            end)

            if not ok then
                dropMovers()
            end

            return ok
        end

        local function flyStep()
            if S.tpState then
                return
            end

            local root = getRoot()
            if not root then
                return
            end

            if not velocity or velocity.Parent ~= root then
                if not attach(root) then
                    return
                end
            end

            local camera = Workspace.CurrentCamera
            if not camera then
                return
            end

            local humanoid = humanoidOf()
            if humanoid and not humanoid.PlatformStand then
                humanoid.PlatformStand = true
            end

            local frame = camera.CFrame
            local direction = Vector3.zero

            -- PC keys (IY style)
            if not UserInputService:GetFocusedTextBox() then
                for _, entry in ipairs(FLY_KEYS) do
                    if UserInputService:IsKeyDown(entry.key) then
                        if entry.axis == "look" then
                            direction += frame.LookVector * entry.sign
                        elseif entry.axis == "right" then
                            direction += frame.RightVector * entry.sign
                        else
                            direction += frame.UpVector * entry.sign
                        end
                    end
                end
            end

            -- Mobile / thumbstick: Humanoid.MoveDirection (same approach as Infinite Yield mobile)
            if humanoid then
                local md = humanoid.MoveDirection
                if md.Magnitude > 0.05 then
                    local look = Vector3.new(frame.LookVector.X, 0, frame.LookVector.Z)
                    local right = Vector3.new(frame.RightVector.X, 0, frame.RightVector.Z)
                    if look.Magnitude > 0.05 then
                        look = look.Unit
                    else
                        look = Vector3.new(0, 0, -1)
                    end
                    if right.Magnitude > 0.05 then
                        right = right.Unit
                    else
                        right = Vector3.new(1, 0, 0)
                    end
                    -- Project move direction onto camera-relative ground plane + keep Y from keys
                    local flatMd = Vector3.new(md.X, 0, md.Z)
                    if flatMd.Magnitude > 0.05 then
                        direction += flatMd.Unit
                    end
                    -- If player is using jump on mobile while flying, lift a bit
                    if UserInputService.TouchEnabled and humanoid.Jump then
                        direction += frame.UpVector
                    end
                end
            end

            if direction.Magnitude > 0.1 then
                velocity.Velocity = direction.Unit * flySpeed
            else
                velocity.Velocity = Vector3.zero
            end

            if gyro then
                gyro.CFrame = frame
            end
        end

        local function noclipStep()
            local character = LocalPlayer.Character
            if not character then
                return
            end

            for _, part in ipairs(character:GetDescendants()) do
                if part:IsA("BasePart") and part.CanCollide then
                    if collisions[part] == nil then
                        collisions[part] = true
                    end

                    part.CanCollide = false
                end
            end
        end

        function Move.setFly(value)
            flyActive = value

            if value then
                local root = getRoot()
                if root then
                    attach(root)
                end

                if not flyConn then
                    flyConn = RunService.Heartbeat:Connect(function()
                        if not flyActive then
                            return
                        end

                        local ok, err = pcall(flyStep)
                        if not ok then
                            reportError("fly", err)
                        end
                    end)
                end

                return
            end

            if flyConn then
                flyConn:Disconnect()
                flyConn = nil
            end

            dropMovers()

            local humanoid = humanoidOf()
            if humanoid then
                pcall(function()
                    humanoid.PlatformStand = false
                    humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                end)
            end
        end

        function Move.setFlySpeed(value)
            flySpeed = math.clamp(value, 10, 500)
        end

        function Move.setNoclip(value)
            noclipActive = value

            if value then
                if not noclipConn then
                    noclipConn = RunService.Heartbeat:Connect(function()
                        if not noclipActive then
                            return
                        end

                        local ok, err = pcall(noclipStep)
                        if not ok then
                            reportError("noclip", err)
                        end
                    end)
                end

                return
            end

            if noclipConn then
                noclipConn:Disconnect()
                noclipConn = nil
            end

            for part, state in pairs(collisions) do
                if part.Parent then
                    pcall(function()
                        part.CanCollide = state
                    end)
                end
            end

                        if not jumpActive then

                            pcall(function()
                                humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
                            end)

                    end)
                end

            end

            if jumpConn then
                jumpConn:Disconnect()

            end
        end

            Move.setNoclip(false)
            Move.setInfJump(false)

        end

do
    local function install()
        local GLIDE_SPEED = 350
        local AIM_RATE = 9
        local SNAP_GAP = 0.35
        local RESPONSE = 200
        local STREAM_GAP = 0.5
        local HOLD_FORCE = 1e7
        local HOLD_TORQUE = 1e7

        local attachment
        local mover
        local aligner
        local cursor
        local facing
        local goalFrame
        local aimSpot
        local streamClock = 0
        local glideConn
        local running = false

        local function humanoidOf()
            local character = LocalPlayer.Character
            return character and character:FindFirstChildOfClass("Humanoid")
        end

        local function detach()
            if mover then
                pcall(function()
                    mover:Destroy()
                end)
                mover = nil
            end

            if aligner then
                pcall(function()
                    aligner:Destroy()
                end)
                aligner = nil
            end

            if attachment then
                pcall(function()
                    attachment:Destroy()
                end)
                attachment = nil
            end
        end

        local function attach(root)
            detach()

            local ok = pcall(function()
                local point = Instance.new("Attachment")
                point.Name = "FayyMengGlidePoint"
                point.Parent = root

                local position = Instance.new("AlignPosition")
                position.Name = "FayyMengGlidePosition"
                position.Mode = Enum.PositionAlignmentMode.OneAttachment
                position.Attachment0 = point
                position.RigidityEnabled = false
                position.ApplyAtCenterOfMass = true
                position.ReactionForceEnabled = false
                position.MaxForce = HOLD_FORCE
                position.MaxVelocity = math.huge
                position.Responsiveness = RESPONSE
                position.Position = root.Position
                position.Parent = root

                local orientation = Instance.new("AlignOrientation")
                orientation.Name = "FayyMengGlideOrientation"
                orientation.Mode = Enum.OrientationAlignmentMode.OneAttachment
                orientation.Attachment0 = point
                orientation.RigidityEnabled = false
                orientation.ReactionTorqueEnabled = false
                orientation.MaxTorque = HOLD_TORQUE
                orientation.MaxAngularVelocity = math.huge
                orientation.Responsiveness = RESPONSE
                orientation.CFrame = root.CFrame.Rotation
                orientation.Parent = root

                attachment = point
                mover = position
                aligner = orientation
            end)

            if not ok then
                detach()
            end

            return ok
        end

        local function step(deltaTime)
            if not goalFrame then
                return
            end

            local root = getRoot()
            if not root then
                return
            end

            if not mover or mover.Parent ~= root or not aligner or aligner.Parent ~= root then
                if not attach(root) then
                    return
                end

                cursor = root.Position
                facing = root.CFrame.Rotation
            end

            local humanoid = humanoidOf()
            if humanoid and not humanoid.PlatformStand then
                pcall(function()
                    humanoid.PlatformStand = true
                end)
            end

            cursor = cursor or root.Position
            facing = facing or root.CFrame.Rotation

            local delta = goalFrame.Position - cursor
            local span = GLIDE_SPEED * deltaTime

            if delta.Magnitude <= math.max(span, SNAP_GAP) then
                cursor = goalFrame.Position
            else
                cursor += delta.Unit * span
            end

            streamClock += deltaTime
            if streamClock >= STREAM_GAP then
                streamClock = 0
                requestStream(goalFrame.Position)
            end

            local look = goalFrame.Rotation

            if aimSpot then
                local gap = aimSpot - cursor
                if gap.Magnitude > 0.1 then
                    look = CFrame.lookAt(cursor, aimSpot).Rotation
                end
            end

            facing = facing:Lerp(look, 1 - math.exp(-AIM_RATE * deltaTime))

            mover.Position = cursor
            aligner.CFrame = facing
        end

        function Move.glide(goal, aim)
            if typeof(goal) ~= "CFrame" then
                return false
            end

            goalFrame = goal
            aimSpot = typeof(aim) == "Vector3" and aim or nil

            if not running then
                running = true

                local root = getRoot()
                if root then
                    cursor = root.Position
                    facing = root.CFrame.Rotation
                    attach(root)
                end
            end

            if not glideConn then
                glideConn = RunService.Heartbeat:Connect(function(deltaTime)
                    if not running then
                        return
                    end

                    local ok, err = pcall(step, deltaTime)
                    if not ok then
                        reportError("glide", err)
                    end
                end)
            end

            return true
        end

        function Move.glideStop()
            running = false
            goalFrame = nil
            aimSpot = nil
            cursor = nil
            facing = nil
            streamClock = 0

            if glideConn then
                glideConn:Disconnect()
                glideConn = nil
            end

            detach()

            local root = getRoot()
            if root then
                pcall(function()
                    root.AssemblyLinearVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero
                end)
            end

            local humanoid = humanoidOf()
            if humanoid then
                pcall(function()
                    humanoid.PlatformStand = false
                    humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                end)
            end
        end
    end

    install()
end


local function mountainPeakPosition()
    local cx = Workspace:GetAttribute("MountainCenterX")
    local cz = Workspace:GetAttribute("MountainCenterZ")
    local peakY = Workspace:GetAttribute("MountainPeakY")
    local baseY = Workspace:GetAttribute("MountainBaseY")
    local radius = Workspace:GetAttribute("MountainRadius")

    if typeof(cx) ~= "number" or typeof(cz) ~= "number" then
        local spot = mountainSpot()
        if not spot then
            return nil
        end
        cx, cz = spot.X, spot.Z
    end
    if typeof(radius) ~= "number" or radius < 20 then
        radius = 150
    end
    if typeof(peakY) ~= "number" then
        peakY = (typeof(baseY) == "number" and baseY or 0) + 900
    end
    if typeof(baseY) ~= "number" then
        baseY = 0
    end

    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    local char = LocalPlayer.Character
    if char then
        rayParams.FilterDescendantsInstances = { char }
    end
    rayParams.IgnoreWater = true

    local bestPos = Vector3.new(cx, peakY, cz)
    local bestY = -math.huge
    local samples = {
        Vector3.new(0, 0, 0),
        Vector3.new(1, 0, 0), Vector3.new(-1, 0, 0),
        Vector3.new(0, 0, 1), Vector3.new(0, 0, -1),
        Vector3.new(0.7, 0, 0.7), Vector3.new(-0.7, 0, 0.7),
        Vector3.new(0.7, 0, -0.7), Vector3.new(-0.7, 0, -0.7),
    }
    local rings = { 0, radius * 0.05, radius * 0.1, radius * 0.18, radius * 0.28 }

    for _, ring in ipairs(rings) do
        for _, dir in ipairs(samples) do
            local ox = cx + dir.X * ring
            local oz = cz + dir.Z * ring
            local origin = Vector3.new(ox, peakY + 250, oz)
            local hit = Workspace:Raycast(origin, Vector3.new(0, -(peakY - baseY + 400), 0), rayParams)
            if hit and hit.Position.Y > bestY then
                bestY = hit.Position.Y
                bestPos = hit.Position
            end
        end
    end

    if bestY < -1e8 then
        bestPos = Vector3.new(cx, peakY, cz)
    end

    return bestPos + Vector3.new(0, 6, 0)
end

local function teleportToMountainPeak()
    local peak = mountainPeakPosition()
    if not peak then
        WindUI:Notify({
            Title = "Mine A Mountain",
            Content = "Mountain peak not found",
            Duration = 3,
            Icon = "alert-circle",
        })
        return false
    end

    requestStream(peak)
    pcall(function()
        LocalPlayer:RequestStreamAroundAsync(peak, 5)
    end)

    if not Move or type(Move.glide) ~= "function" then
        WindUI:Notify({
            Title = "Mine A Mountain",
            Content = "Tween system unavailable",
            Duration = 3,
            Icon = "alert-circle",
        })
        return false
    end

    pcall(function()
        if Move.setNoclip then
            Move.setNoclip(true)
        end
    end)
    pcall(function()
        if Move.glideStop then
            Move.glideStop()
        end
    end)

    S._peakTweening = true
    task.spawn(function()
        local deadline = os.clock() + 45
        local attempts = 0
        while S._peakTweening and os.clock() < deadline do
            local goal = mountainPeakPosition() or peak
            peak = goal
            requestStream(goal)
            pcall(function()
                LocalPlayer:RequestStreamAroundAsync(goal, 5)
            end)
            pcall(function()
                if Move.setNoclip then Move.setNoclip(true) end
            end)
            pcall(function()
                Move.glide(CFrame.new(goal), goal)
            end)
            attempts += 1

            local arrived = false
            for _ = 1, 25 do
                if not S._peakTweening then
                    break
                end
                local root = getRoot()
                if root and (root.Position - goal).Magnitude <= 14 then
                    -- re-scan true peak; continue if higher ground found
                    local recheck = mountainPeakPosition()
                    if recheck and recheck.Y > goal.Y + 8 and (root.Position - recheck).Magnitude > 18 then
                        peak = recheck
                        break
                    end
                    arrived = true
                    break
                end
                task.wait(0.2)
            end

            if arrived then
                break
            end
            task.wait(0.15)
        end

        S._peakTweening = false
        pcall(function()
            if Move and Move.glideStop then Move.glideStop() end
        end)
        pcall(function()
            if Move and Move.setNoclip then Move.setNoclip(false) end
        end)
        local root = getRoot()
        local humanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if root then
            pcall(function()
                root.Anchored = false
                root.AssemblyLinearVelocity = Vector3.zero
                root.AssemblyAngularVelocity = Vector3.zero
            end)
        end
        if humanoid then
            pcall(function()
                humanoid.PlatformStand = false
                if humanoid.WalkSpeed < 8 then humanoid.WalkSpeed = 16 end
            end)
        end

        local root2 = getRoot()
        local finalPeak = mountainPeakPosition() or peak
        local dist = root2 and (root2.Position - finalPeak).Magnitude or 999
        WindUI:Notify({
            Title = "Mine A Mountain",
            Content = dist <= 18
                and string.format("At peak Y=%.0f", finalPeak.Y)
                or string.format("Peak attempt done (%.0fm away)", dist),
            Duration = 3,
            Icon = dist <= 18 and "check" or "alert-circle",
        })
    end)

    WindUI:Notify({
        Title = "Mine A Mountain",
        Content = "Tweening to mountain peak...",
        Duration = 2,
        Icon = "check",
    })
    return true
end

local Net = {}
local netConns = {}

do
    local function install()
        local HttpService = game:GetService("HttpService")
        local TeleportService = game:GetService("TeleportService")
        local GuiService = game:GetService("GuiService")
        local PLACE = game.PlaceId
        local PAGES = 1
        local POOL_TARGET = 20
        local RETRY_STEP = 1.5
        local RETRY_LIMIT = 10
        local BACK_DELAY = 3
        local REFILL_MARK = 8
        local WARM_STEP = 30

        local visited = {}
        local pool = {}
        local hopping = false
        local reviving = false
        local lastCode = 0
        local alive = true

        local function note(text)
            pcall(function()
                WindUI:Notify({
                    Title = "Hop",
                    Content = text,
                    Duration = 4,
                    Icon = "globe",
                })
            end)
        end

        local function grab(link)
            local sender = (syn and syn.request) or (http and http.request) or http_request or request

            if type(sender) == "function" then
                local ok, response = pcall(function()
                    return sender({ Url = link, Method = "GET" })
                end)

                if ok and type(response) == "table" then
                    local code = tonumber(response.StatusCode) or 0

                    if code >= 200 and code < 300 and type(response.Body) == "string" then
                        return response.Body, code
                    end

                    return nil, code
                end
            end

            local ok, body = pcall(function()
                return game:HttpGet(link)
            end)

            if ok and type(body) == "string" then
                return body, 200
            end

            return nil, 0
        end

        local function fetch(cursor)
            local link = string.format("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Desc&excludeFullGames=true&limit=100", PLACE)

            if type(cursor) == "string" and cursor ~= "" then
                link ..= "&cursor=" .. cursor
            end

            local body, code = grab(link)

            if type(body) ~= "string" then
                lastCode = code or 0
                return nil, nil
            end

            lastCode = 200

            local parsed, data = pcall(function()
                return HttpService:JSONDecode(body)
            end)

            if not parsed or type(data) ~= "table" then
                lastCode = -1
                return nil, nil
            end

            return data.data, data.nextPageCursor
        end

        local function refill()
            table.clear(pool)

            local cursor

            for _ = 1, PAGES do
                local entries, nextCursor = fetch(cursor)

                if type(entries) == "table" then
                    for _, entry in ipairs(entries) do
                        local id = type(entry) == "table" and entry.id or nil

                        if type(id) == "string" and id ~= game.JobId and not visited[id] then
                            local playing = tonumber(entry.playing) or 0
                            local room = tonumber(entry.maxPlayers) or 0

                            if room == 0 or playing < room then
                                pool[#pool + 1] = id
                            end
                        end
                    end
                end

                cursor = nextCursor

                if type(cursor) ~= "string" or cursor == "" or #pool >= POOL_TARGET then
                    break
                end
            end

            for index = #pool, 2, -1 do
                local swap = math.random(1, index)
                pool[index], pool[swap] = pool[swap], pool[index]
            end

            return #pool
        end

        function Net.rejoin()
            if reviving then
                return
            end

            reviving = true
            local jobId = game.JobId

            task.spawn(function()
                for _ = 1, RETRY_LIMIT do
                    local sent = pcall(function()
                        TeleportService:TeleportToPlaceInstance(PLACE, jobId, LocalPlayer)
                    end)

                        local sent, err = pcall(function()
                            TeleportService:TeleportToPlaceInstance(PLACE, choice, LocalPlayer)
                        end)

                        if not sent then
                            note("blocked  " .. tostring(err))
                        end

                        if #pool <= REFILL_MARK then
                            task.spawn(refill)
                        end
                    else
                        note(string.format("no servers  http %d", lastCode))
                    end

                    task.wait(RETRY_STEP)
                end

                hopping = false
            end)

            return true
        end

        function Net.forget()
            table.clear(visited)
            table.clear(pool)
            visited[game.JobId] = true
        end

        function Net.ready()

        end

local function resolveUserId(input)
    if input == nil then
        return nil
    end
    input = tostring(input):gsub("%s+", "")
    if input == "" then
        return nil
    end
    local asNum = tonumber(input)
    if asNum and asNum > 0 then
        return math.floor(asNum)
    end
    local ok, userId = pcall(function()
        return Players:GetUserIdFromNameAsync(input)
    end)
    if ok and type(userId) == "number" and userId > 0 then
        return userId
    end
    return nil
end

local function joinServerByUser(input)
    local userId = resolveUserId(input)
    if not userId then
        WindUI:Notify({
            Title = "Join Server",
            Content = "Invalid username / userId",
            Duration = 3,
            Icon = "alert-circle",
        })
        return
    end

    task.spawn(function()
        local ts = game:GetService("TeleportService")
        local placeId, jobId
        local errMsg

        -- IMPORTANT: capture multi-return inside pcall (pcall only keeps 1st return)
        local ok = pcall(function()
            local a, b, c = ts:GetPlayerPlaceInstanceAsync(userId)
            -- API variants across Roblox versions / executors:
            -- (placeId, jobId)
            -- (success, placeId, jobId)
            -- (jobId) only
            if type(a) == "boolean" then
                if not a then
                    errMsg = tostring(b)
                    error(errMsg or "not in game")
                end
                placeId, jobId = b, c
            elseif type(a) == "number" or (type(a) == "string" and tonumber(a)) then
                placeId, jobId = tonumber(a) or a, b
            elseif type(a) == "string" and type(b) == "string" then
                -- some return jobId first
                if #a > 10 and not tonumber(a) then
                    jobId, placeId = a, tonumber(b) or game.PlaceId
                else
                    placeId, jobId = tonumber(a) or a, b
                end
            else
                placeId, jobId = a, b
            end
        end)

        if not ok or not jobId then
            -- HTTP fallback via presence (same game only)
            local httpOk, data = pcall(function()
                local req = (syn and syn.request) or (http and http.request) or request or http_request
                if not req then
                    error("no request")
                end
                local res = req({
                    Url = "https://apis.roblox.com/presence-api/v1/presence/users",
                    Method = "POST",
                    Headers = { ["Content-Type"] = "application/json" },
                    Body = game:GetService("HttpService"):JSONEncode({ userIds = { userId } }),
                })
                return game:GetService("HttpService"):JSONDecode(res.Body)
            end)

            if httpOk and data and data.userPresences then
                for _, presence in ipairs(data.userPresences) do
                    if presence.userPresenceType == 2 and presence.placeId and presence.gameId then
                        placeId = presence.placeId
                        jobId = presence.gameId
                        break
                    end
                end
            end
        end

        placeId = tonumber(placeId) or placeId
        jobId = jobId and tostring(jobId) or nil

        if not jobId or jobId == "" then
            WindUI:Notify({
                Title = "Join Server",
                Content = errMsg or "Player offline / privacy blocked",
                Duration = 3,
                Icon = "alert-circle",
            })
            return
        end

        if placeId and tonumber(placeId) and tonumber(placeId) ~= game.PlaceId then
            WindUI:Notify({
                Title = "Join Server",
                Content = "They are in a different place (" .. tostring(placeId) .. ")",
                Duration = 3,
                Icon = "alert-circle",
            })
            return
        end

        if jobId == game.JobId then
            WindUI:Notify({
                Title = "Join Server",
                Content = "Already in the same server",
                Duration = 2,
                Icon = "check",
            })
            return
        end

        WindUI:Notify({
            Title = "Join Server",
            Content = "Joining server...",
            Duration = 3,
            Icon = "globe",
        })

        local tpOk, tpErr = pcall(function()
            ts:TeleportToPlaceInstance(game.PlaceId, jobId, LocalPlayer)
        end)
        if not tpOk then
            pcall(function()
                ts:TeleportToPlaceInstance(tonumber(placeId) or game.PlaceId, jobId, LocalPlayer)
            end)
        end
    end)
end

local function nearestOtherPlayer(origin)
    local best, bestDist = nil, math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local char = plr.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            if hrp then
                local d = (hrp.Position - origin).Magnitude
                if d < bestDist then
                    bestDist = d
                    best = hrp
                end
            end
        end
    end
    return best, bestDist
end

local function safeModeStep()
    if not S.safeMode then
        if S.safeFleeing then
            S.safeFleeing = false
            S.safeReturnPos = nil
        end
        return false
    end

    -- Only while auto farm / money / mining boulder
    local farming = S.farmActive or S.miningBoulder ~= nil
    if not farming and not S.safeFleeing then
        return false
    end

    local root = getRoot()
    if not root then
        return false
    end

    local threat, dist = nearestOtherPlayer(root.Position)
    local SAFE_RADIUS = 200
    local FLEE_DIST = 1000

    if S.safeFleeing then
        -- Return when clear
        if not threat or dist >= SAFE_RADIUS + 50 then
            local ret = S.safeReturnPos
            S.safeFleeing = false
            S.safeReturnPos = nil
            if ret then
                if Move and Move.glide then
                    pcall(Move.glide, CFrame.new(ret), ret)
                    task.delay(2.5, function()
                        if Move and Move.glideStop then
                            pcall(Move.glideStop)
                        end
                    end)
                else
                    pcall(teleportTo, ret)
                end
            end
            return false
        end
        return true -- still fleeing, pause farm
    end

    if threat and dist <= SAFE_RADIUS then
        S.safeReturnPos = root.Position
        S.safeFleeing = true
        -- Flee away from threat
        local away = root.Position - threat.Position
        if away.Magnitude < 1 then
            away = Vector3.new(1, 0, 0)
        end
        away = Vector3.new(away.X, 0, away.Z).Unit
        local fleePos = root.Position + away * FLEE_DIST + Vector3.new(0, 40, 0)
        if Move and Move.glide then
            pcall(Move.setNoclip, true)
            pcall(Move.glide, CFrame.new(fleePos), fleePos)
        else
            pcall(teleportTo, fleePos)
        end
        return true
    end

    return false
end

            for _, lift in ipairs(AIM_LIFT) do
                for _, angle in ipairs(AIM_ANGLES) do
                    local dir = (CFrame.fromAxisAngle(Vector3.yAxis, math.rad(angle)) * away).Unit
                    local candidate = center + dir * reach + Vector3.new(0, lift, 0)

                    if sightClear(candidate, part, model) then
                        if skip <= 0 then
                            return CFrame.new(candidate, center)
                        end

                        skip -= 1
                    end
                end
            end

            return CFrame.new(center + away * reach + Vector3.new(0, AIM_LIFT[2], 0), center)
        end

        local function step(deltaTime)
            local root = getRoot()
            if not root then
                statusText = "Waiting for character"
                return

            if safeModeStep() then

            if phase == "scan" then
                if not scanned then

                    requestStream(spot)
                    pcall(function()
                        LocalPlayer:RequestStreamAroundAsync(spot, 3)
                    end)
                    scanned = true
                    noTargetClock = 0
                    statusText = "Scanning targets"
                end

                local model = pickTarget()
                if not model then
                    noTargetClock = (noTargetClock or 0) + deltaTime
                    local spot = mountainSpot() or root.Position
                    if noTargetClock < 12 then
                        if math.floor(noTargetClock * 2) % 2 == 0 then
                            requestStream(spot)
                        end
                        statusText = string.format("Waiting for boulders %.0fs", noTargetClock)
                        return
                    end
                    beginLoot(true)
                    statusText = "Final rune sweep"
                    return
                end

                noTargetClock = 0
                target = model
                anchor = visibleAnchor(model)
                spotFrame = nil
                aimTurn = 0
                blindClock = 0
                lostClock = 0
                dryClock = 0
                probeIndex = 0
                hpMark = nil
                dryRounds = 0
                phase = "mine"
                return
            end

            if phase == "mine" then
                if not target or not target.Parent then
                    S.miningBoulder = nil
                    lastSpot = root.CFrame
                    beginLoot(false)
                    return
                end

                S.miningBoulder = target

                local kind = boulderKind(target) or "Boulder"
                local hp = boulderHealth(target)

                if hp and hp <= 0 then
                    lastSpot = root.CFrame
                    beginLoot(false)
                    return
                end

                if not anchor or not anchor.Parent or not anchor:IsDescendantOf(target) then
                    anchor = visibleAnchor(target) or anchorOf(target)
                    spotFrame = nil
                end

                local core = coreSpot(target)
                local center = core or anchorSpot(target, anchor)

                if not center then
                    lostClock += deltaTime

                    if lostClock >= LOST_GRACE then
                        lastSpot = root.CFrame
                        beginLoot(false)
                        return
                    end

                    statusText = string.format("Holding %s", kind)
                    return
                end

                lostClock = 0

                if not spotFrame then
                    if core and probeIndex == 0 then
                        spotFrame = coreFrame(core, anchor)
                    else
                        spotFrame = approach(anchor, target, aimTurn, center, PROBE_DIST[probeIndex % #PROBE_DIST + 1])
                    end
                end

                if spotFrame then
                    Move.setNoclip(true)
                    hold(spotFrame, center)
                    statusText = string.format("Mining %s", kind)
                end

                if heldPick == nil or heldPick.Parent ~= LocalPlayer.Character then
                    equipClock = 0
                    heldPick = equipPick()
                else
                    equipClock += deltaTime
                    if equipClock >= EQUIP_STEP then
                        equipClock = 0
                        heldPick = equipPick() or heldPick
                    end
                end

                if not heldPick then
                    statusText = "No pickaxe"
                    return
                end

                swingClock += deltaTime

                local gap = swingGap()
                local swung = 0

                if swingClock >= gap then
                    swung = math.min(math.floor(swingClock / gap), 4)
                    swingClock -= swung * gap

                    for _ = 1, swung do
                        swing(anchor, target, center)
                    end
                end

                if hp then
                    if hpMark == nil or hp < hpMark - 0.001 then
                        hpMark = hp
                        dryRounds = 0
                        dryClock = 0
                        blindClock = 0
                    else
                        dryClock += deltaTime
                        dryRounds += swung

                        if dryClock >= DRY_TIME or dryRounds >= DRY_ROUNDS then
                            dryClock = 0
                            dryRounds = 0
                            probeIndex += 1
                            aimTurn = (aimTurn + 1) % #AIM_ANGLES
                            anchor = freshAnchor(target) or visibleAnchor(target) or anchor
                            spotFrame = nil
                        end
                    end

                    statusText = string.format("Mining %s  %.0f hp", kind, hp)
                    return
                end

                if sightClear(root.Position, anchor, target) then
                    blindClock = 0
                else
                    blindClock += deltaTime

                    if blindClock >= SIGHT_GRACE then
                        blindClock = 0
                        aimTurn = (aimTurn + 1) % #AIM_ANGLES
                        anchor = visibleAnchor(target) or anchor
                        spotFrame = nil
                    end
                end

                statusText = string.format("Mining %s", kind)

                return
            end

            if phase == "loot" then
                if lastSpot then
                    hold(lastSpot)
                end

                Mountain.grabNear(RUNE_SWEEP)

                if now < lootUntil then
                    statusText = string.format("Looting runes  %.1fs", lootUntil - now)
                    return
                end

                if pendingFinish then
                    phase = "reset"
                    waitUntil = now + RESET_WAIT
                    statusText = "Runes collected"
                else
                    target = nil
                    anchor = nil
                    phase = "scan"
                    statusText = "Next boulder"
                end

                return
            end

            if phase == "reset" then
                if now < waitUntil then
                    return
                end

                statusText = "Restarting server"
                WindUI:Notify({
                    Title = "FayyMeng",
                    Content = "No boulders left - kicking and rejoining",
                    Duration = 3,
                    Icon = "alert-circle",
                })
                stop()
                restart()
            end
        end

            end
        end

            target = nil

            lastSpot = nil
            swingClock = 0
            equipClock = 0
            hpMark = nil
            dryRounds = 0
            spotFrame = nil
            aimTurn = 0

            scanIndex = 0
            heldPick = nil
            pendingFinish = false
            lootUntil = 0
            statusText = "Starting"

        end

        FarmBox:Toggle({

        })

                end
            end

                labelClock = 0

            end
        end)

        Farm.stop = stop
        Farm.equipPick = equipPick
        Farm.swingGap = swingGap
        Farm.digEvent = digEvent
    end

    install()
end

                    statusText = string.format("Breaking crystal %d%%", math.floor(ratio * 100))
                else
                    statusText = "Collecting crystals"
                end

                return
            end

            local origin = farmOrigin(root)

            if target and now - surfaceClock >= SURFACE_GAP then
                surfaceClock = now

                local spot = surfaceAt(target.X, target.Z)

                if not spot then
                    target = nil
                    columnY = nil
                    columnDry = 0
                    columnSwings = 0
                else
                    if not columnY or spot.Y < columnY - 0.05 then
                        columnDry = 0
                    else
                        columnDry += columnSwings
                    end

                    columnSwings = 0
                    columnY = spot.Y
                    target = spot

                    if columnDry >= COLUMN_DRY then
                        target = nil
                        columnY = nil
                        columnDry = 0
                    end
                end
            end

            if not target then
                local spot = pickTarget(origin, now)

                if not spot then
                    spot = surfaceAt(origin.X, origin.Z)
                end

                if not spot then
                    requestStream(origin)

                    if canSwing then
                        swingClock -= swingNeed
                        swing(root.Position - Vector3.new(0, DIG_REACH * 0.5, 0))
                    end

                    statusText = "Loading terrain"
                    return
                end

                target = spot
                columnY = spot.Y
                columnDry = 0
                columnSwings = 0
                surfaceClock = now
            end

            holdAt(CFrame.new(target + Vector3.new(0, DIG_LIFT, 0), target), target)

            if canSwing then
                swingClock -= swingNeed
                columnSwings += 1
                swing(target)
            end

            statusText = string.format("Mining surface at %dm", math.floor(target.Y))
        end

        local function setActive(value)
            if not value then
                stop()
                return
            end

            active = true
            S.farmActive = true
            target = nil
            columnY = nil
            columnDry = 0
            columnSwings = 0
            surfaceClock = 0
            peakClock = 0
            scanIndex = 0
            scanUntil = 0
            loaded = false

do
    local swingClock = 0

    local function getHeldPick()
        local char = LocalPlayer.Character
        if not char then
            return nil
        end
        local tool = char:FindFirstChildOfClass("Tool")
        if tool and (getAttr(tool, "IsPickaxe") == true or getAttr(tool, "DigPower") ~= nil) then
            return tool
        end
        return nil
    end

    local miningCheckClock = 0
    local isMiningCrystal = false
    local MINING_CHECK_GAP = 0.35
    local MINING_RANGE = 18

    local SWING_BURST = 6
    local DIG_RANGE = 20

    local miningParams = OverlapParams.new()
    miningParams.FilterType = Enum.RaycastFilterType.Exclude
    pcall(function()
        miningParams.MaxParts = 40
        miningParams.RespectCanCollide = false
    end)

    local aimParams = RaycastParams.new()
    aimParams.FilterType = Enum.RaycastFilterType.Exclude
    aimParams.IgnoreWater = true

    local function refreshMiningState(root)
        local now = os.clock()
        if now - miningCheckClock < MINING_CHECK_GAP then
            return isMiningCrystal
        end
        miningCheckClock = now

        miningParams.FilterDescendantsInstances = { LocalPlayer.Character or LocalPlayer }

        local ok, hits = pcall(function()
            return Workspace:GetPartBoundsInRadius(root.Position, MINING_RANGE, miningParams)
        end)

        if not ok or not hits then
            isMiningCrystal = false
            return false
        end

        for _, part in ipairs(hits) do
            if part and part.Parent then
                local hp = getAttr(part, "MinedHP")
                if hp == nil and part.Parent then
                    hp = getAttr(part.Parent, "MinedHP")
                end
                if hp ~= nil then
                    local num = tonumber(hp)
                    if num == nil or num > 0 then
                        isMiningCrystal = true
                        return true
                    end
                end
            end
        end

        isMiningCrystal = false
        return false
    end

    local function getLookDigAim(camera, character)
        local origin = camera.CFrame.Position
        local look = camera.CFrame.LookVector

        aimParams.FilterDescendantsInstances = { character }
        local hit = Workspace:Raycast(origin, look * (DIG_RANGE + 25), aimParams)

        if hit and hit.Position then
            local dist = (hit.Position - origin).Magnitude
            if dist <= DIG_RANGE + 10 then
                return hit.Position
            end
            return origin + look * DIG_RANGE
        end

        return origin + look * DIG_RANGE
    end

    local function fastDigStep(deltaTime)
        if not S.fastDigActive then
            return
        end

        local pick = getHeldPick()
        if not pick then
            return
        end

        local event = Farm and Farm.digEvent and Farm.digEvent()
        if not event then
            return
        end

        local root = getRoot()
        local camera = Workspace.CurrentCamera
        if not root or not camera then
            return
        end

        local character = LocalPlayer.Character
        if not character then
            return
        end

        if refreshMiningState(root) then
            return
        end

        if isAtBase() or not isInMountainZone(root.Position) then
            swingClock = 0
            return
        end

        local aim = getLookDigAim(camera, character)
        if not aim then
            return
        end

        if isInNoDigZone(aim) then
            return
        end

        local mult = tonumber(S.digSpeedMult) or 0.4
        mult = math.clamp(mult, 0.05, 1)
        local baseGap = (Farm.swingGap and Farm.swingGap(pick)) or 0.04
        local swingNeed = math.max(0.02, baseGap * mult)

        swingClock = swingClock + deltaTime
        if swingClock < swingNeed then
            return
        end

        swingClock = swingClock - swingNeed
        if swingClock > swingNeed then
            swingClock = 0
        end

        local name = pick.Name
        local cf = camera.CFrame
        local look = cf.LookVector
        local right = cf.RightVector
        local up = Vector3.yAxis

        local flat = Vector3.new(look.X, 0, look.Z)
        if flat.Magnitude < 0.05 then
            flat = Vector3.new(right.X, 0, right.Z)
        end
        flat = flat.Unit
        local side = Vector3.new(-flat.Z, 0, flat.X)

        local r = 4
        pcall(function()
            event:FireServer(name, aim)
            event:FireServer(name, aim + flat * r)
            event:FireServer(name, aim - flat * r)
            event:FireServer(name, aim + side * r)
            event:FireServer(name, aim - side * r)
            event:FireServer(name, aim + up * r)
            event:FireServer(name, aim + look * 1.5)
        end)
    end

    local function setFastDig(value)
        S.fastDigActive = value == true
        swingClock = 0
    end

    local function install()
        local FastDigBox = Tabs.dig

        FastDigBox:Toggle({
            Title = "Fast Dig",
            Value = false,
            Callback = setFastDig,
        })

    end

    install()

    fastDigConn = RunService.Heartbeat:Connect(function(deltaTime)
        local ok, err = pcall(fastDigStep, deltaTime)
        if not ok then
            reportError("fastDig", err)
        end
    end)
end

                return math.max(1, math.floor(value))

            if type(value) == "string" then
                local n = tonumber(value:match("[%d%.]+"))
                if n and n > 0 then
                    return math.floor(n)

            if type(value) == "table" then
                local v = value.Text or value.Value or value[1] or value.Input
                return parseAmount(v)

            return math.max(1, math.floor(tonumber(placeAmount) or 1))

                    local ly = oh.Y + 0.05
                    local bottomLocal = Vector3.new(lx, ly - half.Y, lz)
                    return cf:PointToWorldSpace(bottomLocal)

        local function runeWorldPosition(inst)
            if inst:IsA("BasePart") then
                return inst.Position

            if inst:IsA("Model") then
                local ok, pivot = pcall(function()
                    return inst:GetPivot().Position
                end)
                if ok and pivot then
                    return pivot
                end
                local part = inst.PrimaryPart or inst:FindFirstChild("Handle") or inst:FindFirstChildWhichIsA("BasePart", true)
                if part then

                end
            end
            return nil

        local function runeStillOnPlot(rune, plot)
            if not rune or not rune.Parent then
                return false

            if plot and rune:IsDescendantOf(plot) then
                return true

            if getAttr(rune, "IsRune") == true or (type(rune.Name) == "string" and rune.Name:find("Rune", 1, true)) then
                return rune.Parent ~= nil
            end
            return false

        local function firePickupRune(Event, rune)
            pcall(function()

                pcall(function()
                    if typeof(fireproximityprompt) == "function" then
                        fireproximityprompt(prompt)
                    end

                pcall(function()
                    prompt:InputHoldBegin()
                end)
                task.wait(0.05)
                pcall(function()
                    prompt:InputHoldEnd()
                end)
            end
        end

        local function waitUntilPicked(rune, plot, timeout)
            local deadline = os.clock() + (timeout or 1.2)
            while os.clock() < deadline do
                if not runeStillOnPlot(rune, plot) then
                    return true
                end
                task.wait(0.08)
            end
            return not runeStillOnPlot(rune, plot)
        end

                    else

                    local root = getRoot()
                    if root then
                        pcall(function()
                            root.CFrame = CFrame.new(centerPos)
                        end)

                end

                if Move and Move.setFly then
                    pcall(Move.setFly, false)
                end
                if Move and Move.setNoclip then
                    pcall(Move.setNoclip, false)
                end

                WindUI:Notify({
                    Title = "Pickup Runes",
                    Content = string.format("Success %d | Failed %d | at plot center", picked, failed),
                    Duration = 3,
                    Icon = picked > 0 and "check" or "alert-circle",
                })
            end)
        end

        local function buyAllAvailableBombs()
            local query = BombShopQuery
            local buyReq = BombBuyRequest
            if not query or not buyReq then
                return 0
            end
            local ok, data = pcall(function()
                return query:InvokeServer()
            end)
            if not ok or not data or not data.stock then
                return 0
            end
            local boughtTotal = 0
            for bombId, amount in pairs(data.stock) do
                local count = tonumber(amount) or 0
                if count > 0 then
                    for _ = 1, count do
                        local success, res = pcall(function()
                            return buyReq:InvokeServer(bombId)
                        end)
                        if success and res and res.ok then
                            boughtTotal += 1
                        else
                            break
                        end
                    end
                end
            end
            return boughtTotal
        end

local function setAntiRagdoll(value)
    S.antiRagdoll = value == true

    if S._antiRagdollConn then
        S._antiRagdollConn:Disconnect()
        S._antiRagdollConn = nil
    end
    if S._antiRagdollAttrConn then
        S._antiRagdollAttrConn:Disconnect()
        S._antiRagdollAttrConn = nil
    end
    if S._antiRagdollCharConn then
        S._antiRagdollCharConn:Disconnect()
        S._antiRagdollCharConn = nil
    end
    if S._antiRagdollStateConn then
        S._antiRagdollStateConn:Disconnect()
        S._antiRagdollStateConn = nil
    end

    if not S.antiRagdoll then
        return
    end

    local function getDeactivateRemote()
        local pkg = ReplicatedStorage:FindFirstChild("RagdollSystemPackage")
        local sys = pkg and pkg:FindFirstChild("RagdollSystem")
        local remotes = sys and sys:FindFirstChild("Remotes")
        return remotes and remotes:FindFirstChild("DeactivateRagdollRemote")
    end

    local function forceUnragdoll()
        local character = LocalPlayer.Character
        if not character then
            return
        end

        local remote = getDeactivateRemote()
        if remote then
            pcall(function()
                remote:FireServer()
            end)
        end

        -- also try main remotes folder variant
        local alt = findRemote and findRemote("RagdollRequest")
        if alt then
            pcall(function()
                alt:FireServer("disable")
            end)
            pcall(function()
                alt:FireServer("deactivate")
            end)
        end

        pcall(function()
            character:SetAttribute("Ragdolled", false)
        end)

        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local root = character:FindFirstChild("HumanoidRootPart")
        if humanoid then
            pcall(function()
                humanoid.PlatformStand = false
                if humanoid.WalkSpeed <= 0 then
                    humanoid.WalkSpeed = 16
                end
                if humanoid.JumpPower <= 0 and humanoid.JumpHeight ~= nil then
                    -- leave jump as-is if game uses JumpHeight
                end
                local state = humanoid:GetState()
                if state == Enum.HumanoidStateType.Physics
                    or state == Enum.HumanoidStateType.FallingDown
                    or state == Enum.HumanoidStateType.Ragdoll then
                    humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                end
            end)
        end

        if root then
            pcall(function()
                root.AssemblyAngularVelocity = Vector3.zero
            end)
        end

        -- re-enable animation constraints if movement ragdoll disabled them
        for _, inst in ipairs(character:GetDescendants()) do
            if inst:IsA("AnimationConstraint") then
                pcall(function()
                    inst.Enabled = true
                    inst.IsKinematic = true
                end)
            end
        end
    end

    local function hookCharacter(character)
        if S._antiRagdollAttrConn then
            S._antiRagdollAttrConn:Disconnect()
            S._antiRagdollAttrConn = nil
        end
        if not character then
            return
        end

        S._antiRagdollAttrConn = character:GetAttributeChangedSignal("Ragdolled"):Connect(function()
            if S.antiRagdoll and character:GetAttribute("Ragdolled") then
                forceUnragdoll()
            end
        end)

        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            if S._antiRagdollStateConn then
                S._antiRagdollStateConn:Disconnect()
            end
            S._antiRagdollStateConn = humanoid.StateChanged:Connect(function(_, newState)
                if not S.antiRagdoll then
                    return
                end
                if newState == Enum.HumanoidStateType.Physics
                    or newState == Enum.HumanoidStateType.Ragdoll
                    or newState == Enum.HumanoidStateType.FallingDown then
                    forceUnragdoll()
                end
            end)
        end

        if character:GetAttribute("Ragdolled") then
            forceUnragdoll()
        end
    end

    hookCharacter(LocalPlayer.Character)
    S._antiRagdollCharConn = LocalPlayer.CharacterAdded:Connect(hookCharacter)

    local clock = 0
    S._antiRagdollConn = RunService.Heartbeat:Connect(function(dt)
        if not S.antiRagdoll then
            return
        end
        clock += dt
        if clock < 0.05 then
            return
        end
        clock = 0

        local character = LocalPlayer.Character
        if not character then
            return
        end

        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local ragdolled = character:GetAttribute("Ragdolled") == true
        local physics = humanoid and (
            humanoid:GetState() == Enum.HumanoidStateType.Physics
            or humanoid:GetState() == Enum.HumanoidStateType.Ragdoll
            or humanoid.PlatformStand == true
        )

        if ragdolled or physics then
            forceUnragdoll()
        end
    end)
end

do
    local ADMIN_NAMES = {
        Firemalte07 = true,
        BAiLeymen = true,
    }

    local panicked = false

    local function isAdminPlayer(player)
        if not player then
            return false
        end
        if ADMIN_NAMES[player.Name] then
            return true
        end
        if ADMIN_NAMES[player.DisplayName] then
            return true
        end
        return false
    end

    local function adminInGame()
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and isAdminPlayer(player) then
                return true, player
            end
        end
        return false, nil
    end

    local function silentPanic()
        if panicked then
            return
        end
        panicked = true

        pcall(function()
            S.espActive = false
            S.playerEspActive = false
            S.luckEspActive = false
            S.autoPickupActive = false
            S.pickupValueActive = false
            S.pickupLuckActive = false
            S.fastDigActive = false
            S.aimTpEnabled = false
            S.instantPromptActive = false
            S.speedActive = false
        end)

        pcall(function()
            if Farm and Farm.stop then Farm.stop() end
        end)
        pcall(function()
            if Money and Money.stop then Money.stop() end
        end)
        pcall(function()
            if Mountain and Mountain.shutdown then Mountain.shutdown() end
        end)
        pcall(function()
            if Move and Move.shutdown then Move.shutdown() end
        end)
        pcall(function()
            if Move and Move.setFly then Move.setFly(false) end
        end)
        pcall(function()
            if Move and Move.setNoclip then Move.setNoclip(false) end
        end)

        pcall(function()
            if clearEsp then clearEsp() end
        end)
        pcall(function()
            if clearPlayerEsp then clearPlayerEsp() end
        end)
        pcall(function()
            if clearLuckEsp then clearLuckEsp() end
        end)
        pcall(function()
            if EspHolder then
                EspHolder:ClearAllChildren()
            end
        end)

        pcall(function()
            if Window and Window.Destroy then
                Window:Destroy()
            end
        end)
        pcall(function()
            if EspHolder then
                EspHolder:Destroy()
            end
        end)
    end

    local function scanAdmins()
        local found = adminInGame()
        if found then
            silentPanic()
        end
    end

    task.defer(scanAdmins)

    pcall(function()
        Players.PlayerAdded:Connect(function(player)
            if isAdminPlayer(player) then
                silentPanic()
            end
        end)
    end)

    task.spawn(function()
        while not panicked do
            task.wait(3)
            if panicked then
                break
            end
            scanAdmins()
        end
    end)
end
